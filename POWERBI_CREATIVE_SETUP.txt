STEP 2: CREATE ADVANCED MEASURES (30 minutes)
================================================================================

--- TIER 1: CORE KPI MEASURES ---

Total Crashes =
COUNTA(cleaned_accidents_data[crash_date])

Fatal Crashes =
SUM(cleaned_accidents_data[injuries_fatal])

Total Injuries =
SUM(cleaned_accidents_data[injuries_total])

Severe Injuries =
SUM(cleaned_accidents_data[injuries_incapacitating])

--- TIER 2: RATE & PERCENTAGE MEASURES ---

Fatal Rate % =
DIVIDE(SUM(cleaned_accidents_data[injuries_fatal]), COUNTA(cleaned_accidents_data[crash_date]), 0) * 100

Injury Rate % =
DIVIDE(SUM(cleaned_accidents_data[injuries_total]), COUNTA(cleaned_accidents_data[crash_date]), 0) * 100

Severe Rate % =
DIVIDE(SUM(cleaned_accidents_data[injuries_incapacitating]), SUM(cleaned_accidents_data[injuries_total]), 0) * 100

--- TIER 3: ADVANCED ANALYTICS ---

Avg Units per Crash =
AVERAGE(cleaned_accidents_data[num_units])

Avg Injuries per Crash =
AVERAGE(cleaned_accidents_data[injuries_total])

Crashes with Injuries =
CALCULATE(COUNTA(cleaned_accidents_data[crash_date]), cleaned_accidents_data[injuries_total] > 0)

Injury-Free Crashes =
CALCULATE(COUNTA(cleaned_accidents_data[crash_date]), cleaned_accidents_data[injuries_total] = 0)

Injury-Free Rate % =
DIVIDE(CALCULATE(COUNTA(cleaned_accidents_data[crash_date]), cleaned_accidents_data[injuries_total] = 0), COUNTA(cleaned_accidents_data[crash_date]), 0) * 100

--- TIER 4: RISK INDICATORS ---

High Risk Indicator =
IF(DIVIDE(SUM(cleaned_accidents_data[injuries_fatal]), COUNTA(cleaned_accidents_data[crash_date]), 0) > 0.002, "🔴 HIGH RISK", IF(DIVIDE(SUM(cleaned_accidents_data[injuries_fatal]), COUNTA(cleaned_accidents_data[crash_date]), 0) > 0.001, "🟠 MEDIUM RISK", "🟢 LOW RISK"))

Peak Hour Crashes =
MAXX(GROUPBY(cleaned_accidents_data, cleaned_accidents_data[crash_hour], "Count", COUNTA(cleaned_accidents_data[crash_date])), [Count])

✓ Verify: All 15 measures created
